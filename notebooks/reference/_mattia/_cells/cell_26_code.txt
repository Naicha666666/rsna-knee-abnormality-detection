def _ke_sha256(_ke_path):
    _ke_digest = _ke_hashlib.sha256()
    with _KePath(_ke_path).open('rb') as _ke_handle:
        for _ke_block in iter(lambda: _ke_handle.read(8 << 20), b''):
            _ke_digest.update(_ke_block)
    return _ke_digest.hexdigest()
